        };
      }

      init() {
        if (!this.ctx) {
          const AudioContext = window.AudioContext || window.webkitAudioContext;
          this.ctx = new AudioContext();
          this.masterGain = this.ctx.createGain();
          this.applyGain();
          this.masterGain.connect(this.ctx.destination);
        }
        if (this.ctx.state === 'suspended') {
          this.ctx.resume();
        }
      }

      applyGain() {
        if (!this.masterGain || !this.ctx) return;
        const effectiveVol = this.isMuted ? 0 : this.currentVolume * 0.35;
        this.masterGain.gain.setValueAtTime(effectiveVol, this.ctx.currentTime);
      }

      setVolume(percent) {
        this.init();
        this.currentVolume = Math.max(0, Math.min(100, percent)) / 100;
        if (this.currentVolume > 0 && this.isMuted) {
          this.isMuted = false;
        }
        this.applyGain();
        this.updateUI();
      }

      toggleMute() {
        this.init();
        this.isMuted = !this.isMuted;
        if (!this.isMuted && this.currentVolume === 0) {
          this.currentVolume = 0.3;
        }
        this.applyGain();
        this.updateUI();
      }

      updateUI() {
        const indicator = document.getElementById('vol-indicator');
        const muteBtn = document.getElementById('btn-vol-mute');
        const slider = document.getElementById('vol-slider');
        if (!indicator || !muteBtn || !slider) return;

          indicator.style.color = "#ff0055";
          muteBtn.innerText = "🔇";
          muteBtn.classList.add('muted');
          slider.value = 0;
        } else {
          const pct = Math.round(this.currentVolume * 100);
          indicator.innerText = `${pct}%`;
          indicator.style.color = "var(--text-accent)";
          muteBtn.innerText = "🔊";
          muteBtn.classList.remove('muted');
          slider.value = pct;
        }
      }

      playDropSound(isHard = false) {
        if (!this.ctx || this.isMuted || this.currentVolume === 0) return;
        try {
          const now = this.ctx.currentTime;
          const osc = this.ctx.createOscillator();
          const gain = this.ctx.createGain();

          osc.type = isHard ? 'triangle' : 'square';
          const startFreq = isHard ? 120 : 200;
          const endFreq = 40;

          osc.frequency.setValueAtTime(startFreq, now);
          osc.frequency.exponentialRampToValueAtTime(endFreq, now + (isHard ? 0.12 : 0.06));

          gain.gain.setValueAtTime(isHard ? 0.45 : 0.2, now);
          gain.gain.linearRampToValueAtTime(0.01, now + (isHard ? 0.12 : 0.06));

          osc.connect(gain);
          gain.connect(this.masterGain);

          osc.start(now);
          osc.stop(now + (isHard ? 0.12 : 0.06));
        } catch (e) {}
      }

      startBGM() {
        this.init();
        if (this.isPlayingBGM) return;
        this.isPlayingBGM = true;
        this.currentStep = 0;
        this.scheduleNextNote();
      }

      stopBGM() {
        this.isPlayingBGM = false;
        if (this.bgmTimer) {
          clearTimeout(this.bgmTimer);
          this.bgmTimer = null;
        }
      }

      updateTempoByHeight(highestRowRatio) {
        const targetBpm = this.baseBpm + Math.round(highestRowRatio * 90);
        this.currentBpm = Math.min(230, Math.max(130, targetBpm));
      }

      scheduleNextNote() {
        if (!this.isPlayingBGM || !this.ctx) return;

        const noteItem = this.melody[this.currentStep];
        const noteName = noteItem[0];
        const durationUnits = noteItem[1];
        
        const beatSec = 60 / this.currentBpm;
        const durationSec = durationUnits * beatSec;

        if (noteName && !this.isMuted && this.currentVolume > 0) {
          const freq = this.freqMap[noteName];
          if (freq) {
            const now = this.ctx.currentTime;
            const osc = this.ctx.createOscillator();
            const gain = this.ctx.createGain();

            osc.type = 'square';
            osc.frequency.setValueAtTime(freq, now);

            gain.gain.setValueAtTime(0.08, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + durationSec * 0.85);

            osc.connect(gain);
            gain.connect(this.masterGain);

            osc.start(now);
            osc.stop(now + durationSec * 0.88);
          }
        }

        this.currentStep = (this.currentStep + 1) % this.melody.length;
        this.bgmTimer = setTimeout(() => {
          this.scheduleNextNote();
        }, durationSec * 1000);
      }
    }

    let nextPieceQueue = [];

    function createPiece(type) {
      switch (type) {
        case 'T': return [[0, 1, 0], [1, 1, 1], [0, 0, 0]];
        case 'I': return [[0, 2, 0, 0], [0, 2, 0, 0], [0, 2, 0, 0], [0, 2, 0, 0]];
        case 'S': return [[0, 3, 3], [3, 3, 0], [0, 0, 0]];
        case 'Z': return [[4, 4, 0], [0, 4, 4], [0, 0, 0]];
        case 'L': return [[0, 5, 0], [0, 5, 0], [0, 5, 5]];
        case 'O': return [[6, 6], [6, 6]];
        case 'J': return [[0, 7, 0], [0, 7, 0], [7, 7, 0]];
      }
    }

    function getRandomPiece() {
      const pieces = 'ILJOTSZ';
      return createPiece(pieces[Math.floor(pieces.length * Math.random())]);
    }

    function fillNextQueue() {
      while (nextPieceQueue.length < 2) {
        nextPieceQueue.push(getRandomPiece());
      }
    }

    function drawNextPieces() {
      if (!nextContext) return;
      nextContext.clearRect(0, 0, nextCanvas.width, nextCanvas.height);

      nextContext.strokeStyle = 'rgba(255, 255, 255, 0.08)';
      nextContext.lineWidth = 1;
      nextContext.beginPath();
      nextContext.moveTo(42, 4);
      nextContext.lineTo(42, nextCanvas.height - 4);
      nextContext.stroke();

      const size = 8.5;

      nextPieceQueue.slice(0, 2).forEach((piece, idx) => {
        if (!piece) return;
        const centerX = idx === 0 ? 21 : 63;
        const offsetX = centerX - (piece[0].length * size) / 2;
        const offsetY = (nextCanvas.height - piece.length * size) / 2;

        piece.forEach((row, y) => {
          row.forEach((value, x) => {
            if (value !== 0) {
              const item = PALETTE[value];
              if (item) {
                nextContext.fillStyle = item.base;
                nextContext.fillRect(offsetX + x * size + 0.5, offsetY + y * size + 0.5, size - 1, size - 1);
                nextContext.strokeStyle = item.light;
                nextContext.lineWidth = 1;
                nextContext.strokeRect(offsetX + x * size + 0.5, offsetY + y * size + 0.5, size - 1, size - 1);
              }
            }
          });
        });
      });
    }

    function createMatrix(w, h) {
      const matrix = [];
      while (h--) matrix.push(new Array(w).fill(0));
      return matrix;
    }

    let arena = createMatrix(COLS, ROWS);

    const player = {
      pos: { x: 0, y: 0 },
      matrix: null,
      score: 0,
      lines: 0,
      level: 1,
      isPlaying: false
    };

    function collide(arenaTarget, piece) {
      const [m, o] = [piece.matrix, piece.pos];
      for (let y = 0; y < m.length; ++y) {
        for (let x = 0; x < m[y].length; ++x) {
          if (m[y][x] !== 0 &&
              (arenaTarget[y + o.y] && arenaTarget[y + o.y][x + o.x]) !== 0) {
            return true;
          }
        }
      }
      return false;
    }

    function merge(arenaTarget, p) {
      p.matrix.forEach((row, y) => {
        row.forEach((value, x) => {
          if (value !== 0) arenaTarget[y + p.pos.y][x + p.pos.x] = value;
        });
      });
    }

    function rotate(matrix, dir) {
      for (let y = 0; y < matrix.length; ++y) {
        for (let x = 0; x < y; ++x) {
          [matrix[x][y], matrix[y][x]] = [matrix[y][x], matrix[x][y]];
        }
      }
      if (dir > 0) matrix.forEach(row => row.reverse());
      else matrix.reverse();
    }

    function playerRotate(dir) {
      if (!player.isPlaying || isPaused) return;
      const pos = player.pos.x;
      let offset = 1;
      rotate(player.matrix, dir);
      while (collide(arena, player)) {
        player.pos.x += offset;
        offset = -(offset + (offset > 0 ? 1 : -1));
        if (offset > player.matrix[0].length) {
          rotate(player.matrix, -dir);
          player.pos.x = pos;
          return;
        }
      }
      resetLockDelay();
    }

    function playerMove(dir) {
      if (!player.isPlaying || isPaused) return;
      player.pos.x += dir;
      if (collide(arena, player)) {
        player.pos.x -= dir;
      } else {
        resetLockDelay();
      }
    }

    let lockTimer = 0;
    const LOCK_DELAY = 450;
    let lockResets = 0;
    const MAX_LOCK_RESETS = 12;

    function isGrounded() {
      player.pos.y++;
      const grounded = collide(arena, player);
      player.pos.y--;
      return grounded;
    }

    function resetLockDelay() {
      if (isGrounded() && lockResets < MAX_LOCK_RESETS) {
        lockTimer = 0;
        lockResets++;
      }
    }

    function updateStackTension() {
      let highestOccupiedRow = ROWS;
      for (let y = 0; y < ROWS; y++) {
        if (arena[y].some(val => val !== 0)) {
          highestOccupiedRow = y;
          break;
        }
      }
      const occupiedRows = ROWS - highestOccupiedRow;
      const stackRatio = Math.max(0, Math.min(1, occupiedRows / (ROWS * 0.75)));
      soundEngine.updateTempoByHeight(stackRatio);
    }

    function lockPiece() {
      merge(arena, player);
      soundEngine.playDropSound(false);
      arenaSweep();
      updateStackTension();
      playerReset();
      lockTimer = 0;
      lockResets = 0;
    }

    function playerDrop() {
      if (!player.isPlaying || isPaused) return false;
      player.pos.y++;
      if (collide(arena, player)) {
        player.pos.y--;
        return false;
      }
      player.score += 1;
      updateScore();
      dropCounter = 0;
      return true;
    }

    function playerHardDrop() {
      if (!player.isPlaying || isPaused) return;
      let dropDistance = 0;
      while (!collide(arena, player)) {
        player.pos.y++;
        dropDistance++;
      }
      player.pos.y--;
      dropDistance--;
      player.score += dropDistance * 2;
      updateScore();
      
      soundEngine.playDropSound(true);
      merge(arena, player);
      arenaSweep();
      updateStackTension();
      playerReset();
      lockTimer = 0;
      lockResets = 0;
    }

    function getGhostPosition() {
      const ghost = {
        pos: { x: player.pos.x, y: player.pos.y },
        matrix: player.matrix
      };
      while (!collide(arena, ghost)) {
        ghost.pos.y++;
      }
      ghost.pos.y--;
      return ghost;
    }

    function arenaSweep() {
      let rowCount = 0;
      outer: for (let y = arena.length - 1; y >= 0; --y) {
        for (let x = 0; x < arena[y].length; ++x) {
          if (arena[y][x] === 0) continue outer;
        }
        const rowColors = arena[y].map(val => (PALETTE[val] ? PALETTE[val].base : '#fff'));
        spawnParticles(y, rowColors);

        const row = arena.splice(y, 1)[0].fill(0);
        arena.unshift(row);
        ++y;
        rowCount++;
      }

      if (rowCount > 0) {
        const lineScores = [0, 100, 300, 500, 800];
        player.score += (lineScores[rowCount] || 1000) * player.level;
        player.lines += rowCount;
        player.level = startLevel + Math.floor(player.lines / 10);
        dropInterval = Math.max(80, 800 - (player.level - 1) * 70);
        updateScore();
      }
    }

    function playerReset() {
      fillNextQueue();
      player.matrix = nextPieceQueue.shift();
      fillNextQueue();
      drawNextPieces();

      player.pos.y = 0;
      player.pos.x = Math.floor(COLS / 2) - Math.floor(player.matrix[0].length / 2);

      if (collide(arena, player)) {
        triggerGameOver();
      }
    }

    function triggerGameOver() {
      player.isPlaying = false;
      isPaused = false;
      soundEngine.stopBGM();
      
      const overlay = document.getElementById('game-overlay');
      const msg = document.getElementById('overlay-msg');
      const setupBox = document.getElementById('single-setup-box');
      const scoreSubmitBox = document.getElementById('score-submit-box');
      const pauseBtn = document.getElementById('btn-pause');

      pauseBtn.innerText = "⏸️ 정지";
      overlay.style.display = 'flex';
      msg.innerText = "게임 종료";
      msg.className = "overlay-text lose";

      if (player.score > 0) {
        setupBox.style.display = 'none';
        document.getElementById('score-submit-title').innerText = "🏆 글로벌 랭킹 등록";
        scoreSubmitBox.style.display = 'flex';
        document.getElementById('username-input').focus();
      } else {
        setupBox.style.display = 'flex';
        document.getElementById('overlay-btn').innerText = '다시 시작';
        scoreSubmitBox.style.display = 'none';
      }
    }

    function updateScore() {
      scoreElement.innerText = player.score.toLocaleString();
      linesElement.innerText = player.lines;
      levelElement.innerText = player.level;
    }

    window.startGame = function() {
      if (isPaused) {
        togglePause();
        return;
      }
      soundEngine.startBGM();
      soundEngine.updateTempoByHeight(0);

      startLevel = parseInt(document.getElementById('single-start-level').value, 10) || 1;
      const selectedSize = document.getElementById('single-board-size').value;
      applyBoardSize(selectedSize);

      document.getElementById('game-overlay').style.display = 'none';
      resetGameBoard(startLevel);
      player.isPlaying = true;
      nextPieceQueue = [];
      fillNextQueue();
      playerReset();
    };

    function resetGameBoard(lvl = 1) {
      arena = createMatrix(COLS, ROWS);
      player.score = 0;
      player.lines = 0;
      player.level = lvl;
      dropInterval = Math.max(80, 800 - (player.level - 1) * 70);
      updateScore();
    }

    /* =========================================================
       5. 파티클 이펙트
    ========================================================= */
    const particles = [];
    function spawnParticles(y, colors) {
      for (let x = 0; x < COLS; x++) {
        for (let i = 0; i < 2; i++) {
          particles.push({
            x: (x + 0.5) * BLOCK_W,
            y: (y + 0.5) * BLOCK_H,
            vx: (Math.random() - 0.5) * 5,
            vy: (Math.random() - 0.5) * 5 - 1.5,
            size: Math.random() * 3 + 2,
            color: colors[Math.floor(Math.random() * colors.length)],
            alpha: 1,
            decay: Math.random() * 0.02 + 0.02
          });
        }
      }
    }

    function updateParticles() {
      for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.15;
        p.alpha -= p.decay;
        if (p.alpha <= 0) particles.splice(i, 1);
      }
    }

    function drawParticles(ctx) {
      particles.forEach(p => {
        ctx.save();
        ctx.globalAlpha = p.alpha;
        ctx.fillStyle = p.color;
        ctx.shadowColor = p.color;
        ctx.shadowBlur = 6;
        ctx.fillRect(p.x, p.y, p.size, p.size);
        ctx.restore();
      });
    }

    /* =========================================================
       6. 보드 렌더링
    ========================================================= */
    function drawBlock(ctx, x, y, paletteIdx, isGhost = false) {
      const px = x * BLOCK_W;
      const py = y * BLOCK_H;
      const item = PALETTE[paletteIdx];
      if (!item) return;

      if (isGhost) {
        ctx.save();
        ctx.strokeStyle = item.glow;
        ctx.lineWidth = 1.5;
        ctx.strokeRect(px + 1.5, py + 1.5, BLOCK_W - 3, BLOCK_H - 3);
        ctx.fillStyle = 'rgba(255, 255, 255, 0.03)';
        ctx.fillRect(px + 1.5, py + 1.5, BLOCK_W - 3, BLOCK_H - 3);
        ctx.restore();
        return;
      }

      ctx.save();
      const grad = ctx.createLinearGradient(px, py, px + BLOCK_W, py + BLOCK_H);
      grad.addColorStop(0, item.light);
      grad.addColorStop(0.3, item.base);
      grad.addColorStop(1, '#05020a');

      ctx.fillStyle = grad;
      ctx.fillRect(px + 1, py + 1, BLOCK_W - 2, BLOCK_H - 2);

      ctx.strokeStyle = item.light;
      ctx.lineWidth = 1;
      ctx.strokeRect(px + 2, py + 2, BLOCK_W - 4, BLOCK_H - 4);
      ctx.restore();
    }

    function renderBoard(ctx, boardArena, activePlayer = null) {
      ctx.fillStyle = '#06030e';
      ctx.fillRect(0, 0, canvas.width, canvas.height);

      ctx.strokeStyle = 'rgba(255, 255, 255, 0.04)';
      ctx.lineWidth = 1;
      for (let x = 0; x < COLS; x++) {
        ctx.beginPath();
        ctx.moveTo(x * BLOCK_W, 0);
        ctx.lineTo(x * BLOCK_W, canvas.height);
        ctx.stroke();
      }
      for (let y = 0; y < ROWS; y++) {
        ctx.beginPath();
        ctx.moveTo(0, y * BLOCK_H);
        ctx.lineTo(canvas.width, y * BLOCK_H);
        ctx.stroke();
      }

      boardArena.forEach((row, y) => {
        row.forEach((value, x) => {
          if (value !== 0) drawBlock(ctx, x, y, value);
        });
      });

      if (activePlayer && activePlayer.matrix) {
        if (ghostEnabled) {
          const ghost = getGhostPosition();
          ghost.matrix.forEach((row, y) => {
            row.forEach((value, x) => {
              if (value !== 0) drawBlock(ctx, x + ghost.pos.x, y + ghost.pos.y, value, true);
            });
          });
        }

        activePlayer.matrix.forEach((row, y) => {
          row.forEach((value, x) => {
            if (value !== 0) drawBlock(ctx, x + activePlayer.pos.x, y + activePlayer.pos.y, value);
          });
        });
      }
    }

    function draw() {
      renderBoard(context, arena, player);
      drawParticles(context);
    }

    /* =========================================================
       7. DAS / ARR 입력 엔진
    ========================================================= */
    const keys = {
      ArrowLeft: { pressed: false, timer: 0, lastStep: 0 },
      ArrowRight: { pressed: false, timer: 0, lastStep: 0 },
      ArrowDown: { pressed: false, timer: 0 }
    };

    const DAS = 150;
    const ARR = 40;
    const SOFT_DROP_RATE = 35;

    function handleInput(deltaTime) {
      if (isPaused || !player.isPlaying) return;

      ['ArrowLeft', 'ArrowRight'].forEach(key => {
        const k = keys[key];
        if (k.pressed) {
          k.timer += deltaTime;
          if (k.timer >= DAS) {
            const steps = Math.floor((k.timer - DAS) / ARR);
            if (steps > k.lastStep) {
              const deltaSteps = steps - k.lastStep;
              for (let i = 0; i < deltaSteps; i++) {
                playerMove(key === 'ArrowLeft' ? -1 : 1);
              }
              k.lastStep = steps;
            }
          }
        }
      });

      const down = keys['ArrowDown'];
      if (down.pressed) {
        down.timer += deltaTime;
        if (down.timer >= SOFT_DROP_RATE) {
          playerDrop();
          down.timer = 0;
        }
      }
    }

    window.addEventListener('keydown', event => {
      if (['ArrowLeft', 'ArrowRight', 'ArrowUp', 'ArrowDown', 'Space'].includes(event.code)) {
        if (document.activeElement && document.activeElement.tagName === 'INPUT') return;
        event.preventDefault();
      }

      if (document.activeElement && document.activeElement.tagName === 'INPUT') return;

      if (event.code === 'KeyP' || event.code === 'Escape') {
        togglePause();
        return;
      }

      if (event.repeat || !player.isPlaying || isPaused) return;

      soundEngine.init();

      if (event.code === 'ArrowLeft') {
        keys.ArrowLeft.pressed = true;
        keys.ArrowLeft.timer = 0;
        keys.ArrowLeft.lastStep = 0;
        playerMove(-1);
      } else if (event.code === 'ArrowRight') {
        keys.ArrowRight.pressed = true;
        keys.ArrowRight.timer = 0;
        keys.ArrowRight.lastStep = 0;
        playerMove(1);
      } else if (event.code === 'ArrowDown') {
        keys.ArrowDown.pressed = true;
        keys.ArrowDown.timer = 0;
        playerDrop();
      } else if (event.code === 'ArrowUp') {
        playerRotate(1);
      } else if (event.code === 'Space') {
        playerHardDrop();
      }
    });

    window.addEventListener('keyup', event => {
      if (keys[event.code]) {
        keys[event.code].pressed = false;
        keys[event.code].timer = 0;
        keys[event.code].lastStep = 0;
      }
    });

    function setKeypadState(active) {
      isKeypadActive = active;
      const panel = document.getElementById('touch-bottom-panel');
      const btn = document.getElementById('btn-keypad-toggle');
      if (panel) {
        if (isKeypadActive) panel.classList.add('active');
        else panel.classList.remove('active');
      }
      applyBtnStyle(btn, isKeypadActive);
      updateResponsiveLayout();
    }

    window.toggleVirtualKeypad = () => setKeypadState(!isKeypadActive);

    function setupTouchButton(btnId, action) {
      const el = document.getElementById(btnId);
      if (!el) return;

      const startAction = (e) => {
        e.preventDefault();
        e.stopPropagation();
        soundEngine.init();
        el.classList.add('touched');
        if (!player.isPlaying || isPaused) return;

        if (action === 'left') {
          keys.ArrowLeft.pressed = true;
          keys.ArrowLeft.timer = 0;
          keys.ArrowLeft.lastStep = 0;
          playerMove(-1);
        } else if (action === 'right') {
          keys.ArrowRight.pressed = true;
          keys.ArrowRight.timer = 0;
          keys.ArrowRight.lastStep = 0;
          playerMove(1);
        } else if (action === 'down') {
          keys.ArrowDown.pressed = true;
          keys.ArrowDown.timer = 0;
          playerDrop();
        } else if (action === 'rot') {
          playerRotate(1);
        } else if (action === 'hard') {
          playerHardDrop();
        }
      };

      const stopAction = (e) => {
        e.preventDefault();
        e.stopPropagation();
        el.classList.remove('touched');
        if (action === 'left') {
          keys.ArrowLeft.pressed = false;
          keys.ArrowLeft.timer = 0;
          keys.ArrowLeft.lastStep = 0;
        }
        if (action === 'right') {
          keys.ArrowRight.pressed = false;
          keys.ArrowRight.timer = 0;
          keys.ArrowRight.lastStep = 0;
        }
        if (action === 'down') keys.ArrowDown.pressed = false;
      };

      el.addEventListener('touchstart', startAction, { passive: false });
      el.addEventListener('touchend', stopAction, { passive: false });
      el.addEventListener('touchcancel', stopAction, { passive: false });

      el.addEventListener('mousedown', startAction);
      el.addEventListener('mouseup', stopAction);
      el.addEventListener('mouseleave', stopAction);
    }

    setupTouchButton('v-left', 'left');
    setupTouchButton('v-right', 'right');
    setupTouchButton('v-down', 'down');
    setupTouchButton('v-up', 'rot');
    setupTouchButton('v-drop', 'hard');

    /* =========================================================
       8. 메인 루프 및 초기화
    ========================================================= */
    let dropCounter = 0;
    let dropInterval = 800;
    let lastTime = 0;

    function update(time = 0) {
      const deltaTime = time - lastTime;
      lastTime = time;

      if (player.isPlaying && !isPaused) {
        handleInput(deltaTime);
        updateParticles();

        dropCounter += deltaTime;
        if (dropCounter > dropInterval) {
          playerDrop();
        }

        if (isGrounded()) {
          lockTimer += deltaTime;
          if (lockTimer >= LOCK_DELAY) {
            lockPiece();
          }
        } else {
          lockTimer = 0;
        }
      }

      draw();
      requestAnimationFrame(update);
    }

    const isTouch = ('ontouchstart' in window) || (navigator.maxTouchPoints > 0) || (window.innerWidth <= 768);
    if (isTouch) setKeypadState(true);

    fetchLeaderboard();
    applyBoardSize('standard');
    resetGameBoard(1);
    updateResponsiveLayout();
    soundEngine.updateUI();
    update();
  })();
  </script>
</div>
